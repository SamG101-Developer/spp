# What has no run-time benchmark, and why. Every AST with a Stage11_CodeGen and every module in sources/spp/codegen
# needs a benchmark that covers it (a runtime/ast/<ast> folder, or a manifest's "covers") or a line here; coverage.py
# checks it. One per line: "ast:<name>" or "codegen:<module>", then the reason after "#".

ast:module_prototype                  # walks into its members; emits no code that runs
ast:module_implementation             # walks into its members; emits no code that runs
ast:use_statement                     # an import: emits nothing
ast:use_statement_variable            # an import: emits nothing
ast:type_statement                    # an alias: declares a type, emits no code
ast:type                              # a type: declares, emits no code
ast:generic_parameter                 # resolved before codegen: emits no code
ast:generic_parameter_group           # resolved before codegen: emits no code
ast:class_prototype                   # declares a layout: emits no code
ast:sup_implementation                # walks into its members; emits no code that runs
ast:function_parameter                # part of the prototype; its code is the call's (function_parameter_group)
ast:spread_expression                 # expanded into its tuple's elements before codegen (tuple_literal)
ast:cmp_statement                     # a compile-time constant: emits a global, read like any other value
ast:postfix_expression                # dispatches to its operator, each benchmarked on its own

codegen:llvm_mangle                   # names only
codegen:llvm_artefact                 # writing the object and IR files
codegen:llvm_target                   # target selection
codegen:llvm_ctx                      # the LLVM context and module set-up
codegen:llvm_type                     # LLVM types for S++ types: shapes no instruction on its own
codegen:llvm_layout                   # field layout: shapes no instruction on its own
codegen:llvm_size                     # sizes and alignments: shapes no instruction on its own
codegen:llvm_pass_pipeline            # the optimisation pipeline, which every benchmark runs through
codegen:llvm_pass_cold_path           # branch weights from "cold_path" markers, which no benchmark uses; tests/codegen/check-branch-hints.sh
