FLAMEGRAPH
----------
perf record -g -F 99 -o /tmp/perf.data ./cmake-build-debug-wsl-dev/spp && \
    perf script -i /tmp/perf.data | ../FlameGraph/stackcollapse-perf.pl | ../FlameGraph/flamegraph.pl > flamegraph1.html && \
    rm /tmp/perf.data

CALL TREE
---------
perf record -g -F 99 ./cmake-build-debug-wsl-dev/spp && \
    perf script | c++filt | gprof2dot -f perf --strip --wrap --node-thres=1 --edge-thres=1 | dot -Tsvg -Gsplines=ortho -Grankdir=LR -Gconcentrate=true -o graph.svg && \
    rm ./perf.data

UNIT TESTS
----------
cd tests/test_outputs && \
    ../gtest-parallel/gtest-parallel ../../cmake-build-release-wsl-dev/tests/spp_tests --output_dir=../

CPPCHECK
--------
cppcheck --enable=all --inline-suppr --suppressions-list=.cppcheck-suppressions --check-level=exhaustive --std="c++26" \
    --language=c++ --error-exitcode=1 \
    --template='{file}:{line}:{column}: {severity}: {message} [{id}]' --template-location='{file}:{line}:{column}: note: {info}' \
    --output-file=cppcheck-report.txt \
    -j "$(nproc)" -I headers \
    headers sources main.cpp
